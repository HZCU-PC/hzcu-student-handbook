  var form = document.getElementById('zj-form');
  var txt  = document.getElementById('zj-txt');
  var icon = document.getElementById('zj-icon');
  var box  = document.getElementById('zj-engine');
  var list = document.getElementById('zj-engine-list');
  var btn  = document.getElementById('zj-btn');
  if (!form || !txt || !list) { return; }

  function apply(i) {
    current = i;
    var e = engines[i];
    form.action = e.url;
    txt.name = e.param;
    // 搜索框左侧的图标跟着切换
    if (icon) {
      icon.src = e.icon;
      icon.alt = e.name;
      icon.title = '点击切换搜索引擎（当前：' + e.name + '）';
    }
    if (btn) { btn.title = '使用' + e.name + '搜索'; }
    Array.prototype.forEach.call(list.children, function (li, k) {
      li.className = (k === i) ? 'zj-active' : '';
    });
  }

  engines.forEach(function (e, i) {
    var li = document.createElement('li');
    var img = document.createElement('img');
    img.className = 'zj-engine-icon';
    img.src = e.icon;
    img.alt = e.name;
    li.appendChild(img);
    li.appendChild(document.createTextNode(e.name));
    li.addEventListener('click', function () {
      apply(i);
      box.classList.remove('zj-open');
      txt.focus();
    });
    list.appendChild(li);
  });

  apply(0);

  if (icon) {
    icon.addEventListener('click', function (ev) {
      ev.preventDefault();
      box.classList.toggle('zj-open');
    });
  }

  document.addEventListener('click', function (ev) {
    if (!box.contains(ev.target) && ev.target !== icon) {
      box.classList.remove('zj-open');
    }
  });

  // 调试用：加 ?zj-engine=open 可直接展开下拉，便于检查位置
  if (/[?&]zj-engine=open/.test(location.search)) {
    box.classList.add('zj-open');
  }
  // 调试用：加 ?zj-debug=1 显示布局实测数据（校验用，可随时删除）
  if (/[?&]zj-debug=1/.test(location.search)) {
    setTimeout(function () {
      function cols(sel) {
        var els = document.querySelectorAll(sel);
        if (!els.length) { return sel + ': 0'; }
        var tops = {};
        Array.prototype.forEach.call(els, function (e) {
          tops[Math.round(e.getBoundingClientRect().top)] = 1;
        });
        var rows = Object.keys(tops).length;
        return sel + ': ' + els.length + '个 / ' + rows + '行 = 每行 ' +
          (els.length / rows).toFixed(2) + ' 个, 宽 ' +
          Math.round(els[0].getBoundingClientRect().width) + 'px';

      d.style.cssText = 'position:fixed;left:8px;bottom:8px;z-index:9999;background:#000;color:#0f0;' +

        cols('#zj-engine-list > li');
      document.body.appendChild(d);
    }, 1200);
  }
